bool Zombie:: isValid(int row, int col)
{
	if (row < 0 || col < 0
		|| row >= HEIGHT|| col >= WIDTH)
		return false;

	if (color[row][col])
		return false;

	
	return true;
}
void Zombie::checkforplayer(int tiles[HEIGHT][WIDTH])
{
	int dRow[] = { -1, 0, 1, 0 };
	int dCol[] = { 0, 1, 0, -1 };
	float zombiecx = this->getentity().getGlobalBounds().left + this->getentity().getGlobalBounds().width / 2.0f;
	float zombiecy = this->getentity().getGlobalBounds().top + this->getentity().getGlobalBounds().height / 2.0f;
	for (int i = 0; i < HEIGHT; i++)
		for (int j = 0; j < WIDTH; j++)
			color[i][j] = 0;

	int tilex = static_cast<int>(zombiecx) / 32;
	
	int tiley = static_cast<int>(zombiecy) / 32;
	
	color[tiley][tilex] = 1;
	int reached = 0;
	
	std::queue <std::pair<int, int> > q;
	q.push({tiley,tilex});

	while (!q.empty()) {

		std::pair<int, int> cell = q.front();
		int y = cell.first;
		int x = cell.second;
		q.pop();
		for (int i = 0; i < 4; i++) {
			
			int adjy = y + dRow[i];
			int adjx = x + dCol[i];
			if (abs(tilex -adjx )>3 || abs(tiley-adjy)>3)
			{
				reached = 1;
				break;

			}
			if (isValid( adjy, adjx)) {
				if (tiles[adjy][adjx] == 3)//player detected
				{
					reached = 1;
					//do smth
					this->setpos(adjx * 32, adjy * 32);
					//this->posx = adjx * 32 + 16;
					//this->posy = adjy * 32 + 16;
					
					break;
				}
				q.push({ adjy, adjx });
				color[adjy][adjx] = 1;
			}

		}

		if (reached)
			break;

	}
	

	

}
